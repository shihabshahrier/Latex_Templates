\begin{problembox}{Problem 1: Multiple Regression Diagnostics \& Collinearity [30 Points]}
An empirical study of $N = 120$ regional tech hardware markets for \textbf{Vertex Pro-10} investigates quarterly sales revenue ($Y$, in \$100k) based on Advertising Spend ($X_1$, \$10k), Retail Density ($X_2$, stores/100k hab), Competitor Price Index ($X_3$), and Regional Median Income ($X_4$, \$k).
\end{problembox}

\begin{enumerate}
  \item \textbf{Model Specification \& Matrix Formulation}
  
  Consider the additive linear regression model with an interaction term:
  \begin{equation}
    Y_i = \beta_0 + \beta_1 X_{i1} + \beta_2 X_{i2} + \beta_3 X_{i3} + \beta_4 X_{i4} + \beta_5 (X_{i1} \cdot X_{i3}) + \varepsilon_i, \quad \varepsilon_i \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, \sigma^2)
  \end{equation}
  In vector-matrix form, $\bm{Y} = \bm{X}\bm{\beta} + \bm{\varepsilon}$, where the design matrix $\bm{X} \in \mathbb{R}^{120 \times 6}$ has $i$-th row $\bm{x}_i^\top = \begin{bmatrix} 1 & X_{i1} & X_{i2} & X_{i3} & X_{i4} & X_{i1}X_{i3} \end{bmatrix}$. The OLS estimator and its covariance matrix are:
  \begin{equation}
    \hat{\bm{\beta}} = (\bm{X}^\top \bm{X})^{-1} \bm{X}^\top \bm{Y}, \quad \text{Var}(\hat{\bm{\beta}}) = \sigma^2 (\bm{X}^\top \bm{X})^{-1}
  \end{equation}

\begin{problembox}{Problem 2: Logistic Regression \& Classification Threshold Evaluation [35 Points]}
We analyze customer retention data for fictional SaaS platform \textbf{Synapse Analytics}. The target binary outcome $Y_i \in \{0, 1\}$ represents customer churn ($1 = \text{Churned}, 0 = \text{Retained}$) over a 12-month period across $N = 500$ corporate accounts.
\end{problembox}

\begin{enumerate}
  \item \textbf{Logistic Model Formulation \& Log-Likelihood}
  
  The probability of customer churn $p_i = P(Y_i = 1 \mid \bm{x}_i)$ is modeled using the logit link function:
  \begin{equation}
    \text{logit}(p_i) = \ln \left( \frac{p_i}{1 - p_i} \right) = \bm{x}_i^\top \bm{\beta} = \beta_0 + \beta_1 X_{i1} + \beta_2 X_{i2} + \beta_3 X_{i3} + \beta_4 X_{i4}
  \end{equation}
  where $X_1 =$ Monthly Active Hours, $X_2 =$ Support Ticket Count, $X_3 =$ Contract Tenure (months), and $X_4 =$ Average Session Length. The Bernoulli log-likelihood function maximizes:
  \begin{equation}
    \ell(\bm{\beta}) = \sum_{i=1}^{N} \left[ Y_i \ln(p_i) + (1 - Y_i) \ln(1 - p_i) \right] = \sum_{i=1}^{N} \left[ Y_i \bm{x}_i^\top \bm{\beta} - \ln \left( 1 + \exp(\bm{x}_i^\top \bm{\beta}) \right) \right]
  \end{equation}

  \item \textbf{Python Modeling Code (Statsmodels / Scikit-Learn)}

\begin{lstlisting}[language=Python, caption={Logistic Regression Fitting \& ROC Analysis in Python}]
import pandas as pd, statsmodels.api as sm
from sklearn.metrics import roc_curve, auc

df_synapse = pd.read_csv("synapse_churn.csv")
X = sm.add_constant(df_synapse[['ActiveHours', 'SupportTickets', 'TenureMonths', 'AvgSessionLen']])
y = df_synapse['Churned']
logit_mod = sm.Logit(y, X).fit()
fpr, tpr, _ = roc_curve(y, logit_mod.predict(X))
print(f"Model ROC-AUC Score: {auc(fpr, tpr):.4f}")
\end{lstlisting}

  \item \textbf{Decision Threshold Optimization under Asymmetric Costs}

  In subscription SaaS modeling, customer churn incurs a high replacement cost ($C_{FN} = \$1,200$), whereas preemptive retention incentives cost significantly less ($C_{FP} = \$150$). The expected loss function $\mathcal{L}(\tau)$ for threshold $\tau$ is:
  \begin{equation}
    \mathcal{L}(\tau) = C_{FN} \cdot P(Y=1, \hat{Y}=0 \mid \tau) + C_{FP} \cdot P(Y=0, \hat{Y}=1 \mid \tau)
  \end{equation}

  By lowering the classification cutoff from the default $\tau = 0.50$ to the empirical cost-optimal threshold $\tau^* = 0.38$, Sensitivity improves from $77.1\%$ to $89.3\%$, reducing total quarterly revenue churn risk by an estimated $\$34,200$.
\end{enumerate}

\begin{problembox}{Problem 3: Non-parametric Bootstrap Resampling \& Interval Estimation [35 Points]}
We evaluate latency response times (milliseconds) for \textbf{Nexa Cloud Storage} clusters before and after an infrastructure upgrade. The sample sizes are $n_1 = 45$ (baseline) and $n_2 = 45$ (upgraded). Response times exhibit significant right-skewness, violating standard Gaussian assumptions.
\end{problembox}

\begin{enumerate}
  \item \textbf{Bootstrap Algorithmic Procedure}

  We target estimation of the ratio of median response times $\theta = \frac{\text{Med}(X_{\text{after}})}{\text{Med}(X_{\text{before}})}$. The non-parametric empirical bootstrap procedure executes as follows:
  \begin{enumerate}[label=\roman*.]
    \item Draw $B = 5,000$ bootstrap resamples $\bm{X}_1^{*(b)}$ and $\bm{X}_2^{*(b)}$ with replacement from observed samples.
    \item Compute the bootstrap replicate statistic: $\hat{\theta}^{*(b)} = \frac{\text{Med}(\bm{X}_2^{*(b)})}{\text{Med}(\bm{X}_1^{*(b)})}, \quad b = 1, \dots, B$.
    \item Construct 95\% BCa (Bias-Corrected and Accelerated) confidence intervals to account for skewness and acceleration.
  \end{enumerate}

  \item \textbf{R Parallel Bootstrap Implementation Script}

\begin{lstlisting}[language=R, caption={Non-parametric Percentile \& BCa Bootstrap in R}]
library(boot)

median_ratio_fn <- function(data, indices) {
  d <- data[indices, ]
  return(median(d$latency[d$group == "After"]) / median(d$latency[d$group == "Before"]))
}

nexa_data <- read.csv("nexa_latency.csv")
set.seed(4820)
boot_results <- boot(data = nexa_data, statistic = median_ratio_fn, R = 5000)
boot_ci <- boot.ci(boot_results, type = c("perc", "bca"))
print(boot_ci)
\end{lstlisting}
